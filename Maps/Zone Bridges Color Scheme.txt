### Bridges

#2edb93 - Mixed Types Bridge
#6ebed6 - PC Girder Bridge
#29ffa4 - Portable Steel Bridge
#e8f674 - RC Girder Bridge
#088afa - RC Slab Bridge
#13b7dd - Small Slab Bridge
#b15dec - Steel Girder Bridge
#6470da - Truss Bridge
#03e8f9 - Rigid Frame Bridge
#ffbb00 - PC Box Girder Bridge
#ff8800 - Concrete Arch Bridge
#41df33 - All other

`
CASE 
WHEN "bridge_type"= 'Mixed Types Bridge' THEN '#2edb93'
WHEN "bridge_type"= 'PC Girder Bridge' THEN '#6ebed6'
WHEN "bridge_type"= 'Portable Steel Bridge' THEN '#29ffa4'
WHEN "bridge_type"= 'RC Girder Bridge' THEN '#e8f674'
WHEN "bridge_type"= 'RC Slab Bridge' THEN '#088afa'
WHEN "bridge_type"= 'Small Slab Bridge' THEN '#13b7dd'
WHEN "bridge_type"= 'Steel Girder Bridge' THEN '#b15dec'
WHEN "bridge_type"= 'Truss Bridge' THEN '#6470da'
WHEN "bridge_type"= 'Rigid Frame Bridge' THEN '#03e8f9'
WHEN "bridge_type"= 'PC Box Girder Bridge' THEN '#ffbb00'
WHEN "bridge_type"= 'Concrete Arch Bridge' THEN '#ff8800'
ELSE '#41df33'
END
`

### Culverts

`
CASE
WHEN  strpos( "bridge_type", 'Pipe') THEN '#d7d6d2'
WHEN  strpos( "bridge_type", 'Masonary Arch') THEN '#e8daeb'
WHEN  strpos( "bridge_type", 'Steel Arch') THEN '#d4d9ed'
WHEN strpos( "bridge_type", 'Slab Culvert' ) THEN '#cceef0'
ELSE "f_class_color"
END
